cmake_minimum_required(VERSION 3.16)
project(OpticalBoundaryValidation LANGUAGES CXX)
find_package(Geant4 REQUIRED)
include(${Geant4_USE_FILE})
set(DIAGNOSTICS_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../src/CBDsimOpticalDiagnostics.cc"
    CACHE FILEPATH "Production diagnostics source; override only for before/after regression")
add_executable(optical_boundary_test boundary_test.cc ${DIAGNOSTICS_SOURCE})
target_include_directories(optical_boundary_test PRIVATE ../../include)
target_compile_features(optical_boundary_test PRIVATE cxx_std_17)
target_link_libraries(optical_boundary_test PRIVATE ${Geant4_LIBRARIES})
add_executable(optical_arrival_test arrival_test.cc ${DIAGNOSTICS_SOURCE})
target_include_directories(optical_arrival_test PRIVATE ../../include)
target_compile_features(optical_arrival_test PUBLIC cxx_std_17)
target_link_libraries(optical_arrival_test PRIVATE ${Geant4_LIBRARIES})

add_executable(optical_budget_test budget_test.cc ${DIAGNOSTICS_SOURCE})
target_include_directories(optical_budget_test PRIVATE ../../include)
target_compile_features(optical_budget_test PRIVATE cxx_std_17)
target_link_libraries(optical_budget_test PRIVATE ${Geant4_LIBRARIES})
add_executable(optical_budget_lifecycle_test budget_lifecycle_test.cc ${DIAGNOSTICS_SOURCE})
target_include_directories(optical_budget_lifecycle_test PRIVATE ../../include)
target_compile_features(optical_budget_lifecycle_test PRIVATE cxx_std_17)
target_link_libraries(optical_budget_lifecycle_test PRIVATE ${Geant4_LIBRARIES})
